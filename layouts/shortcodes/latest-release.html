{{/* latest-release banner shortcode Shows every asset in data/assets.yaml whose `date` is within the last 7 days. If nothing was published in the last 7 days, falls back to showing the single most recent asset (so the banner is never empty). Usage: {{< latest-release >}} */}} {{- $all := (index hugo.Data.assets "assets") -}} {{- $now := now -}} {{- $cutoff := $now.AddDate 0 0 -7 -}} {{/* Build a list of {asset, parsedDate} pairs, filtering out entries with no date */}} {{- $withDates := slice -}} {{- range $all -}} {{- $asset := . -}} {{- with .date -}} {{- $t := time . -}} {{- $withDates = $withDates | append (dict "a" $asset "t" $t) -}} {{- end -}} {{- end -}} {{/* Sort newest first */}} {{- $sorted := sort $withDates "t" "desc" -}} {{/* Filter to last 7 days */}} {{- $recent := slice -}} {{- range $sorted -}} {{- if ge .t $cutoff -}}{{- $recent = $recent | append . -}}{{- end -}} {{- end -}} {{/* Fallback to single latest if nothing in the last 7 days */}} {{- $picks := $recent -}} {{- if not $picks -}}{{- $picks = first 1 $sorted -}}{{- end -}} {{- $multi := gt (len $picks) 1 -}} {{- $label := "Latest Release" -}} {{- if $multi -}}{{- $label = printf "This week (%d releases)" (len $picks) -}}{{- end -}} {{- if $picks }}
{{ $label }}
{{- range $picks -}} {{- $item := .a -}} {{- $typeIcon := "✨" -}} {{- if eq $item.type "tool" }}{{- $typeIcon = "🛠️" -}}{{- end -}} {{- if eq $item.type "guide" }}{{- $typeIcon = "📚" -}}{{- end -}} {{- if eq $item.type "video" }}{{- $typeIcon = "📺" -}}{{- end -}} {{- if eq $item.type "snippet" }}{{- $typeIcon = "🧠" -}}{{- end -}} {{- $visBadge := "" -}} {{- if eq $item.visibility "public" }}{{- $visBadge = "🌐 Public" -}}{{- end -}} {{- if eq $item.visibility "ms-internal" }}{{- $visBadge = "🔒 Microsoft" -}}{{- end -}} {{- if eq $item.visibility "private" }}{{- $visBadge = "🔐 Private" -}}{{- end -}} {{- $href := $item.slug | default $item.url -}}
{{ $typeIcon }}
{{ $item.title }}{{ with $item.version }} v{{ . }}{{ end }}
{{ $item.description }}
{{- with $item.date }}📅 {{ dateFormat "Jan 2, 2006" . }}{{ end -}} {{- with $visBadge }}{{ . }}{{ end -}} {{- with $item.tags }}{{- range first 4 . }}#{{ . }}{{ end }}{{ end -}}
{{- end -}}
{{- end -}}
