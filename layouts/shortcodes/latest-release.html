{{/* latest-release banner shortcode Shows every asset in data/assets.yaml whose `date` is within the last 7 days. - 1 item → slim single-line "Latest" band (a signpost, not a hero). - 2+ items → flat card grid labelled "This week · N releases". If nothing was published in the last 7 days, falls back to the single most recent asset rendered as the slim band (so it's never empty). Usage: {{< latest-release >}} */}} {{- $all := (index hugo.Data.assets "assets") -}} {{- $now := now -}} {{- $cutoff := $now.AddDate 0 0 -7 -}} {{/* Build a list of {asset, parsedDate} pairs, filtering out entries with no date */}} {{- $withDates := slice -}} {{- range $all -}} {{- $asset := . -}} {{- with .date -}} {{- $t := time . -}} {{- $withDates = $withDates | append (dict "a" $asset "t" $t) -}} {{- end -}} {{- end -}} {{/* Sort newest first */}} {{- $sorted := sort $withDates "t" "desc" -}} {{/* Filter to last 7 days */}} {{- $recent := slice -}} {{- range $sorted -}} {{- if ge .t $cutoff -}}{{- $recent = $recent | append . -}}{{- end -}} {{- end -}} {{/* Fallback to single latest if nothing in the last 7 days */}} {{- $picks := $recent -}} {{- if not $picks -}}{{- $picks = first 1 $sorted -}}{{- end -}} {{- $multi := gt (len $picks) 1 -}} {{- if $picks }} {{- if $multi }} {{/* ─── 2+ releases this week: flat card grid ─── */}}
This week · {{ len $picks }} releases
{{- range $picks -}} {{- $item := .a -}} {{- $typeIcon := "✨" -}} {{- if eq $item.type "tool" }}{{- $typeIcon = "🛠️" -}}{{- end -}} {{- if eq $item.type "guide" }}{{- $typeIcon = "📚" -}}{{- end -}} {{- if eq $item.type "video" }}{{- $typeIcon = "📺" -}}{{- end -}} {{- if eq $item.type "snippet" }}{{- $typeIcon = "🧠" -}}{{- end -}} {{- $visBadge := "" -}} {{- if eq $item.visibility "public" }}{{- $visBadge = "🌐 Public" -}}{{- end -}} {{- if eq $item.visibility "ms-internal" }}{{- $visBadge = "🔒 Microsoft" -}}{{- end -}} {{- if eq $item.visibility "private" }}{{- $visBadge = "🔐 Private" -}}{{- end -}} {{- $href := $item.slug | default $item.url -}}
{{ $typeIcon }}
{{ $item.title }}{{ with $item.version }} v{{ . }}{{ end }}
{{ $item.description }}
{{- with $item.date }}📅 {{ dateFormat "Jan 2, 2006" . }}{{ end -}} {{- with $visBadge }}{{ . }}{{ end -}}
{{- end -}}
{{- else }} {{/* ─── Single release: slim one-line signpost ─── */}} {{- $item := (index $picks 0).a -}} {{- $href := $item.slug | default $item.url -}} {{- $visBadge := "" -}} {{- if eq $item.visibility "public" }}{{- $visBadge = "Public" -}}{{- end -}} {{- if eq $item.visibility "ms-internal" }}{{- $visBadge = "Microsoft" -}}{{- end -}} {{- if eq $item.visibility "private" }}{{- $visBadge = "Private" -}}{{- end -}} 🆕 Latest {{ $item.title }}{{ with $item.version }} v{{ . }}{{ end }} {{- with $item.date }}{{ dateFormat "Jan 2" . }}{{ end -}} {{- with $visBadge }}{{ . }}{{ end -}} → {{- end }} {{- end -}}
